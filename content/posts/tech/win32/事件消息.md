## Ⅰ.写下winmain入口函数
**Windows应用程序(无控制台)入口为WinMain,定义如下** 
```c++
int __clrcall WinMain(
  [in]           HINSTANCE hInstance,           //当前实例的句柄
  [in, optional] HINSTANCE hPrevInstance,       
  [in]           LPSTR     lpCmdLine,           //应用程序的命令行
  [in]           int       nShowCmd             //控制窗口的显示方式
)
```
## Ⅱ.创建,注册窗口类
**窗口类的结构**
```c++
typedef struct tagWNDCLASSA {
  UINT      style;
  WNDPROC   lpfnWndProc;
  int       cbClsExtra;
  int       cbWndExtra;
  HINSTANCE hInstance;
  HICON     hIcon;
  HCURSOR   hCursor;
  HBRUSH    hbrBackground;
  LPCSTR    lpszMenuName;
  LPCSTR    lpszClassName;
} WNDCLASSA, *PWNDCLASSA, *NPWNDCLASSA, *LPWNDCLASSA;
```
--- 
**示例**
```c++
//类名
TCHAR ClassName[] = TEXT("test");           
WNDCLASS wndclass{};                        //创建类
//填写类结构体参数
//类的成员变量必须全部赋值,因此创建对象时需要初始化,将不重要的值初始化为0
wndclass.lpszClassName=ClassName;           //类名
wndclass.hInstance=hInstance;               //类的句柄,这里填入winmain的参数
wndclass.hbrBackground=(HBRUSH)COLOR_MENU;  //背景颜色
wndclass.lpfnWndProc=WindowProc;            //消息处理函数
//注册窗口类
RegisterClass(&wndclass);
```

## Ⅲ.创建,显示窗口
**函数介绍**
```c++
HWND CreateWindow(
  [in, optional] LPCSTR    lpClassName,
  [in, optional] LPCSTR    lpWindowName,
  [in]           DWORD     dwStyle,
  [in]           int       x,
  [in]           int       y,
  [in]           int       nWidth,
  [in]           int       nHeight,
  [in, optional] HWND      hWndParent,
  [in, optional] HMENU     hMenu,
  [in, optional] HINSTANCE hInstance,
  [in, optional] LPVOID    lpParam
);
```
--- 
**示例**
```c++
//创建窗口
   HWND hwnd = CreateWindow(ClassName,TEXT("test"),WS_OVERLAPPEDWINDOW,10,10,600,300,NULL,NULL,hInstance,NULL);
   if(!hwnd){
      return 0;
   }
   //显示窗口
   ShowWindow(hwnd,SW_SHOW);
```
## Ⅳ.消息循环
```c++
while(GetMessage(&msg,NULL,0,0)){
      //翻译消息
      TranslateMessage(&msg);
      //分发消息
      DispatchMessage(&msg);
   }
   return 0;
```
## Ⅴ.创建消息处理函数
**示例**
```c++
//消息处理函数定义如下,必须严格按照它编写
LRESULT CALLBACK WindowProc(HWND hwnd, UINT uMsg, WPARAM wParam, LPARAM lParam){
  switch(uMsg){
      //窗口被创建
      case WM_CREATE:
         break;
      //窗口销毁
      case WM_DESTROY:
         PostQuitMessage(0);
         break;
      //窗口大小改变
      case WM_SIZE:
         printf("窗口大小改变了\n");
         break;
      //窗口移动
      case WM_MOVE:
         printf("窗口被移动了\n");
         break;
      //负责其它Window,例如按钮的消息
      case WM_COMMAND:
         switch(LOWORD(wParam)){
            case 1:
               ;
               break;
          }
  }
  //没写的消息处理方案,由Windows默认处理,返回值DefultWindowProcess
  return DefWindowProc(hwnd,uMsg,wParam,lParam);
}
```
